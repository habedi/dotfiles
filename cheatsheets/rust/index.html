Rust (stable) Cheatsheet

Quick reference · official docs · std

Hello, World

fn main() {
    println!("Hello, {}!", "world");
}

Run a single file: rustc hello.rs && ./hello. Most projects use Cargo: cargo new app && cd app && cargo run.

Primitive Types

  • Signed: i8 i16 i32 i64 i128 isize
  • Unsigned: u8 u16 u32 u64 u128 usize
  • Float: f32 f64
  • bool, char (4-byte Unicode scalar), () unit
  • Strings: &str (borrowed slice), String (owned, growable)
  • Tuple: (i32, &str) · Array: [T; N] · Slice: &[T]

Control Flow

if x > 0 { ... } else if x == 0 { ... } else { ... }

let sign = if x > 0 { 1 } else { -1 };  // expression

loop { break; }
while i < 10 { i += 1; }

for (i, v) in items.iter().enumerate() { ... }
for n in 0..10 { ... }
for n in (0..=10).rev() { ... }

Match & Patterns

match n {
    0 => println!("zero"),
    1 | 2 => println!("one or two"),
    3..=9 => println!("single digit"),
    x if x < 0 => println!("negative"),
    _ => println!("other"),
}

if let Some(v) = maybe { ... }
let Some(v) = maybe else { return; };
while let Some(item) = it.next() { ... }

Ownership & Borrowing

let s = String::from("hi");
let t = s;          // move; s no longer usable

let a = String::from("hi");
let r = &a;          // shared borrow
println!("{a} {r}");

let mut v = vec![1, 2, 3];
let m = &mut v;     // exclusive borrow
m.push(4);

let c = a.clone();    // deep copy when needed

Rules: at any moment either many shared borrows OR one exclusive borrow, never both.

Functions & Closures

fn add(a: i32, b: i32) -> i32 {
    a + b   // last expression returned
}

fn apply<F: Fn(i32) -> i32>(f: F, x: i32) -> i32 { f(x) }

let sq = |x: i32| x * x;
let inc = |x| x + 1;            // inferred
let capture = || println!("{a}");
let mut counter = || { count += 1; };  // FnMut

    fn new(x: f32, y: f32) -> Self { Self { x, y } }
    fn length(&self) -> f32 {
        (self.x.powi(2) + self.y.powi(2)).sqrt()
    }
}

enum Shape {

Option & Result

fn first(v: &[i32]) -> Option<i32> {
    if v.is_empty() { None } else { Some(v[0]) }
}

fn parse(s: &str) -> Result<i32, std::num::ParseIntError> {
    s.parse::<i32>()
}

let n = parse("42")?;            // propagate
let n = parse("x").unwrap_or(0);
let n = first(&v).map(|x| x + 1);

Traits & Generics

trait Greet {
    fn hello(&self) -> String;
    fn shout(&self) -> String {
        self.hello().to_uppercase()
    }
}

struct En;
impl Greet for En { fn hello(&self) -> String { "hi".into() } }

fn say<G: Greet>(g: &G) { println!("{}", g.hello()); }
fn say_dyn(g: &dyn Greet)  { println!("{}", g.hello()); }
fn make() -> impl Greet { En }

Collections

Iterators

let nums = vec![1, 2, 3, 4, 5];

let sum: i32 = nums.iter().sum();
let doubled: Vec<_> = nums.iter().map(|x| x * 2).collect();
let evens: Vec<_> = nums.iter().filter(|x| *x % 2 == 0).collect();

let first_big = nums.iter().find(|x| **x > 3);
let any_neg   = nums.iter().any(|x| *x < 0);
let all_pos   = nums.iter().all(|x| *x > 0);

let grouped: HashMap<_, Vec<_>> = nums.iter()
    .fold(HashMap::new(), |mut acc, &x| {
        acc.entry(x % 2).or_default().push(x); acc
    });

Error Handling

use std::io::{self, Read};
use std::fs::File;

fn read_all(path: &str) -> io::Result<String> {
    let mut f = File::open(path)?;
    let mut s = String::new();
    f.read_to_string(&mut s)?;
    Ok(s)
}

// custom error with thiserror (or by hand impl Display + Error)
#[derive(Debug, thiserror::Error)]
enum AppError {
    #[error("io: {0}")]    Io(#[from] io::Error),
    #[error("missing")]   Missing,
}

Lifetimes

fn longest<'a>(a: &'a str, b: &'a str) -> &'a str {
    if a.len() > b.len() { a } else { b }
}

struct View<'a> { data: &'a [u8] }

impl<'a> View<'a> {
    fn new(d: &'a [u8]) -> Self { Self { data: d } }
}

Most lifetimes are elided; annotate when the compiler can't relate input and output references.

Smart Pointers

  • Box<T> — heap allocation, single owner.
  • Rc<T> / Arc<T> — reference counted (Arc = atomic, thread-safe).
  • RefCell<T> / Cell<T> — interior mutability (single-threaded).
  • Mutex<T> / RwLock<T> — thread-safe interior mutability.
  • Cow<'a, T> — clone-on-write borrowed/owned.
use std::sync::Arc;
let shared = Arc::new(vec![1, 2, 3]);
let a = shared.clone();

Concurrency

use std::thread;
use std::sync::{Arc, Mutex, mpsc};

let data = Arc::new(Mutex::new(0));
let handles: Vec<_> = (0..4).map(|_| {
    let d = data.clone();
    thread::spawn(move || { *d.lock().unwrap() += 1; })
}).collect();
for h in handles { h.join().unwrap(); }

let (tx, rx) = mpsc::channel();
thread::spawn(move || tx.send("hi").unwrap());
println!("{}", rx.recv().unwrap());

Async / Await

// Cargo.toml: tokio = { version = "1", features = ["full"] }
#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let body = reqwest::get("https://example.com")
        .await?
        .text()
        .await?;
    println!("{}", body.len());
    Ok(())
}

async fn work() -> u32 { 42 }
let (a, b) = tokio::join!(work(), work());

Unsafe

unsafe fn dangerous() { ... }

unsafe {
    let p = &x as *const i32;
    println!("{}", *p);
}

extern "C" {
    fn abs(input: i32) -> i32;
}

Five powers: deref raw pointers, call unsafe fn, access mutable static, implement unsafe trait, access union fields. Borrow checker still applies elsewhere.
