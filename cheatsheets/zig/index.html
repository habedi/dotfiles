Hello, World

const std = @import("std");

pub fn main() !void {
    const stdout = std.io.getStdOut().writer();
    try stdout.print("Hello, {s}!\n", .{"world"});
}

Build & run: zig run hello.zig · Compile: zig build-exe hello.zig

Control Flow

while (i < 10) : (i += 1) { ... }

for (items, 0..) |item, i| {
    std.debug.print("{d}: {}\n", .{ i, item });
}

switch (x) {
    0 => { ... },
    1...9 => { ... },
    else => { ... },
}

Optionals

var maybe: ?u32 = null;
maybe = 7;

if (maybe) |val| {
    std.debug.print("got {}\n", .{val});
} else {
    std.debug.print("none\n", .{});
}

const x = maybe orelse 0;
const y = maybe.?;            // panic on null

Errors

const ParseError = error{ Empty, Invalid };

fn parse(s: []const u8) ParseError!u32 {
    if (s.len == 0) return error.Empty;
    return std.fmt.parseInt(u32, s, 10) catch return error.Invalid;
}

const n = try parse("42");          // propagate
const m = parse("x") catch 0;      // default
parse("x") catch |err| { ... };  // inspect

Functions

fn add(a: i32, b: i32) i32 {
    return a + b;
}

pub fn max(comptime T: type, a: T, b: T) T {
    return if (a > b) a else b;
}

// variadic via anytype
fn log(comptime fmt: []const u8, args: anytype) void {
    std.debug.print(fmt, args);
}

Arrays, Slices, Pointers

const arr = [_]i32{ 1, 2, 3, 4 };  // [4]i32
const slice: []const i32 = arr[1..3];  // {2, 3}
const len = slice.len;

var x: i32 = 10;
const p: *i32 = &x;
p.* = 20;

// many-item pointer
const mp: [*]const u8 = arr[0..].ptr;

Structs & Methods

const Vec2 = struct {
    x: f32,
    y: f32,

    pub fn init(x: f32, y: f32) Vec2 {
        return .{ .x = x, .y = y };
    }

    pub fn length(self: Vec2) f32 {
        return @sqrt(self.x * self.x + self.y * self.y);
    }
};

const v = Vec2.init(3, 4);
const n = v.length();

Allocators

const std = @import("std");

pub fn main() !void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const alloc = gpa.allocator();

    const buf = try alloc.alloc(u8, 1024);
    defer alloc.free(buf);

    var list = std.ArrayList(u32).init(alloc);
    defer list.deinit();
    try list.append(42);
}

Common allocators: GeneralPurposeAllocator, ArenaAllocator, FixedBufferAllocator, page_allocator, c_allocator.

Comptime & Generics

fn List(comptime T: type) type {
    return struct {
        items: []T,
        len: usize,
    };
}

const IntList = List(i32);

comptime {
    const n = 10 * 10;
    @compileLog("computed at compile time", n);
}

Defer & Errdefer

fn work(alloc: std.mem.Allocator) ![]u8 {
    const buf = try alloc.alloc(u8, 64);
    errdefer alloc.free(buf);  // only on error

    try doSomething(buf);
    return buf;                // caller owns it
}

defer file.close();          // always runs on scope exit

Build System (build.zig)

const std = @import("std");

pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});

    const exe = b.addExecutable(.{
        .name = "app",
        .root_source_file = b.path("src/main.zig"),
        .target = target,
        .optimize = optimize,
    });
    b.installArtifact(exe);

    const run = b.addRunArtifact(exe);
    const step = b.step("run", "Run the app");
    step.dependOn(&run.step);
}

Commands: zig build, zig build run, zig build test, zig build -Doptimize=ReleaseFast.

Testing

test "basic add" {
    try testing.expectEqual(4, 2 + 2);

test "slice equal" {

Run with zig test file.zig or zig build test.

Concurrency

// async/await was removed in 0.11; threads are the primary primitive.
const std = @import("std");

fn worker(id: u32) void {
    std.debug.print("worker {}\n", .{id});
}

pub fn main() !void {
    const t = try std.Thread.spawn(.{}, worker, .{1});
    t.join();
}

Sync: std.Thread.Mutex, RwLock, Condition, std.atomic.Value(T).

C Interop

const c = @cImport({
    @cInclude("stdio.h");
});

pub fn main() void {
    _ = c.printf("hello from C\n");
}

Link C: zig build-exe main.zig -lc. Use extern fn to declare external symbols.
